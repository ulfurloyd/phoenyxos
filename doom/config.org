* Keybindings

      :desc "Open Dired"

* Configuration

#+begin_src emacs-lisp
(setq doom-user-dir (expand-file-name "~/Projects/phoenyxos/"))
(setq confirm-kill-emacs nil
      confirm-kill-processes nil)
(after! doom
  (advice-add 'doom/delete-frame-with-prompt :override
              (lambda ()
                (delete-frame))))
#+end_src

* Org Mode

#+begin_src emacs-lisp
(after! org
  (setq org-directory "~/org"
        org-agenda-files '("~/org-agenda")))
#+end_src

#+begin_src emacs-lisp
(add-to-list 'custom-theme-load-path "~/.config/emacs/themes/")
(load-theme 'noctalia t)
